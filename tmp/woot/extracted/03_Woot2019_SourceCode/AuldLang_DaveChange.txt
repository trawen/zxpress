uglisynthstart_ug:  
            di      
            call    screensetup

            ld      hl,auldtitle        ;setup the title
            call    printstring         ;and print it

            ld      a,16+5              ;cyan (set up border colours red/cyan)
            ld      (colour1smc+1),a
            ld      a,2                 ;red
            ld      (colour2smc+1),a
            
            ld      hl,auld0
            call    playsong
            
            ld      a,16+1              ;blue (set up border colours blue/yellow)
            ld      (colour1smc+1),a
            ld      a,6                 ;yellow
            ld      (colour2smc+1),a           
            

            ld      hl,auldtext1        ;setup the first bit of words - should alud
            call    printstring         ;and print them

            ld      hl,auld1            ;setup the song fragment
            call    playsong            ;and play it

 

            ld      hl,auldtext2        ;setup the next bit of words - and never 
            call    printstring         ;and print them

            ld      hl,auld2
            call    playsong


            ld      hl,auldtext3        ;should auld
            call    printstring         

            ld      hl,auld3
            call    playsong

            ld      hl,auldtext4        ;for ths sake
            call    printstring

            ld      hl,auld4
            call    playsong

            ld      hl,auldtext5
            call    printstring

            ld      hl,trill1
            call    playsong
            ld      hl,auld5
            call    playsong
            

            ld      hl,auldtext6
            call    printstring
            ld      hl,trill1
            call    playsong            
            ld      hl,auld6
            call    playsong

            ld      hl,auldtext7
            call    printstring
            ld      hl,trill1
            call    playsong    
            ld      hl,auld7
            call    playsong

            ld      hl,auldtext8
            call    printstring

            ld      hl,auld8
            call    playsong

            ld      a,7           ; 7 is the code for white.
            call    8859          ; set border colour.

            call    wastetime  ;just a little pause

            ld      hl,auldtext9
            call    printstring

            call    wastetime  ;just a little pause
            call    wastetime  ;just a little pause
            call    wastetime  ;just a little pause

            ld      hl,auldtext10
            call    printstring

            ei
            ret      


playsong:   ;called with HL pointing at the musidata
            ld      (songpointer),hl ;prime the songpointer with the song fragment
testloop:
            ld      hl,(songpointer)
            ld      (smc_data+1),hl
            inc     hl
            inc     hl
            ld      (songpointer),hl
smc_data:
            ld      hl,(auld1)
            ld      a,l           ;check if L (low byte) is zero (end of song marker), if so, return
            or      a
            ret     z

;Check for a 'rest' (is the value of H=0)
            ld      a,h
            or      a
            jr      z,rest_routine

            call    playnote

            ;jr      rest_routine  ;THIS IS INSERTING GAPS EACH TIME, FOR TESTING PURPOSES

            jr      testloop         ; Loop to next element in song

;
rest_routine:
            ld      bc,20000         ;insterts an audible gap THIS NEEDS TO BE BASED UPON length of L, ****but isn't, yet***
insert_pause:
            nop
            dec     bc
            ld      a,b
            or      c
            jr      nz,insert_pause
            jr      testloop       ; Loop to next element in song






playnote:
;load data (hl)  l=length, h=pitch
tempo_smc:
            ld      a,4                             ;eg. 4 - wow many times through the loop (sort of tempo multipler)
            ld      (smc_length+1),a                ;setup the length of loop where it's reset
            ld      (noteroll),a ;noteroll          ;and initially

            ld      (notelength),hl

            ld      e,1                             ;make sure there is no errant "previous note" that matches the one playing

jumpbackpoint:                          ;come back here to setup d with the pitch of the note

            ld      hl,noteroll         ;noteroll is a X-to-0 counter that determines how many times
            dec     (hl)                ;it goes through the cycle (i.e. the lenghth in the l register)
            jr      nz,skipcounterdown  ;
smc_length:
            ld      (hl),0              ;reset noteroll to it's value (equates to tempo) (will have been setup previously)


            ld      hl,notelength       ;notelength is length (see also noteroll, though)
            dec     (hl)                ;decrease it until it's zero
            ret     z                   ;


skipcounterdown:

            ld      hl,notepitch        ;d is given the required pitch
            ld      d,(hl)




generateanote:   ;generate a note
                 ;pitch is held in register d
                 ;but first check is it a "new" note ?
            ld      a,d
            cp      e                   ;e holds previous note
            jr      z,sameoldnote       ;if it is the same, then we don't need to change counters etc
newnotebeingplayed:
            xor     a                   ;clear carry flag)
            ld      a,d
            ld      e,a                 ;store that note for comparisson next time
            ld      (smc_ug2+1),a
            rra                         ;a=d/2
            ld      c,a                 ;c is used to keep counter temporarily

sameoldnote:                            ;enter here if if is the same note as before
            ld      a,c                 ;restore counter


            inc     a                   ;inc counter
            ld      b,a                 ;load it into b, for a djnz loop
killsometime:
            nop                          ;4 tstates, one byte
            cp      (hl)                 ;7 tstates, one byte, no purpose - just killing some time - NB can add more time killing here, but notes will need retuning!

            djnz    killsometime
colour1smc:
            ld      a,17                 ;will determine border colour as well.  16+colour.  17=blue
            out     (254),a

            ld      a,d                  ;get the pitch
            sub     c                    ;take away the counter
            inc     a                    ;add 1 to stop it ever being zero
            ld      b,a                  ;put the result into b

killsometime2:      
            nop                          ;4 tstates,one byte
            cp      (hl)                 ;7 tstates, one byte, no purpose - just killing some time

            djnz    killsometime2
colour2smc:
            ld      a,6                  ;also determine border colour.  0+colour .  6=yellow
            out     (254),a

                                         ;now we'll change the pulse width by changing a/c
            ld      a,c

smc_ug1:                                 ;this is smc modified to change whether pulse width is increasing or decreasing
            add     a,255                ;either add or subtract 1  (subtract 1=adding 255)
            ld      c,a                  ;update c

            jp      z,changedirection
smc_ug2:            
            ld      a,0                  ;this is smc modified when a new note is played
            cp      c
            jp      z,changedirection
            jp      jumpbackpoint

changedirection:    
            ld      a,(smc_ug1+1)         ;uses twos complement to turn 1 into 255 and vice versa
            cpl                           ;
            inc     a                     ;
            ld      (smc_ug1+1),a         ;
            jp      jumpbackpoint

            call    5633          ; open channel from rom print routine
            ld      a,56          ; black ink (6) on white paper (0*8) + bright (64*0)

            call    3503          ; clear the screen.

            call    8859          ; set border colour.

printstring: ;cheap and dirty string printing routine, hl points at string (terminated with a 255, rather than 0, to allow easier ctrl codes)
             ld   a,(hl)
             inc  a    ;is it 255
             ret  z    ;if so, exit loop
             dec  a
             rst  16   ;if not, print it
             inc  hl   ;and loop round
             jr   printstring
